  function(gadget_embed_schema name path)
    file(READ "${path}" contents)
    string(LENGTH "${contents}" bytes)
    string(HEX "${contents}" encoded)
    file(APPEND "${gadget_schema_header}" "inline const std::string ${name} = [] {\n  std::string result;\n  result.reserve(${bytes});\n")
    set(offset 0)
    while(offset LESS bytes)
      math(EXPR remaining "${bytes} - ${offset}")
      # C++ guarantees a 65536-character literal representation, including its
      # terminator. Piece size affects embedding only, never the schema domain.
      if(remaining GREATER 65535)
        set(length 65535)
      else()
        set(length ${remaining})
      endif()
      math(EXPR hex_offset "2 * ${offset}")
      math(EXPR hex_length "2 * ${length}")
      string(SUBSTRING "${encoded}" ${hex_offset} ${hex_length} piece)
      string(REGEX REPLACE "(..)" "\\\\x\\1" literal "${piece}")
      file(APPEND "${gadget_schema_header}" "  result.append(\"${literal}\", ${length});\n")
      math(EXPR offset "${offset} + ${length}")
    endwhile()
    file(APPEND "${gadget_schema_header}" "  return result;\n}();\n")
